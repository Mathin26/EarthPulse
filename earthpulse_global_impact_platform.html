RAG + Vector Search Implementation
// EarthPulse.Core/Services/RAGService.cs public class RAGService : IRAGService { private readonly IVectorStore _vectorStore; private readonly IAIService _aiService; // Prompt database — seeded with global impact knowledge private readonly PromptDatabase _promptDb = new() { SystemPrompt = @"You are EarthPulse AI, an expert on global environmental and societal impact. Use ONLY the provided context from the vector store. Cite sources. Be factual.", ImpactFactors = GlobalImpactFactors.All // 500+ seeded facts }; public async Task<RAGResponse> QueryAsync(string userQuery) { // 1. Embed query → float vector (1536-dim) var queryVector = await _aiService.EmbedAsync(userQuery); // 2. Cosine similarity search in vector store var topDocs = await _vectorStore.SearchAsync(queryVector, topK: 5); // 3. Build augmented prompt var prompt = new RAGPrompt { System = _promptDb.SystemPrompt, Context = string.Join("\n---\n", topDocs.Select(d => d.Content)), Query = userQuery }; // 4. Generate response grounded in retrieved context var answer = await _aiService.GenerateAsync(prompt); return new RAGResponse { Answer = answer, Sources = topDocs, Scores = topDocs.Select(d => d.CosineSimilarity).ToList() }; } } // InMemoryVectorStore — O(n) cosine similarity search public class InMemoryVectorStore : IVectorStore { private List<VectorDocument> _store = new(); public Task<List<VectorDocument>> SearchAsync( float[] query, int topK) { return Task.FromResult(_store .Select(doc => { doc.CosineSimilarity = CosineSim(query, doc.Embedding); return doc; }) .OrderByDescending(d => d.CosineSimilarity) .Take(topK) .ToList()); } private static float CosineSim(float[] a, float[] b) { var dot = a.Zip(b, (x,y) => x*y).Sum(); var magA = MathF.Sqrt(a.Sum(x => x*x)); var magB = MathF.Sqrt(b.Sum(x => x*x)); return dot / (magA * magB + 1e-8f); } }
